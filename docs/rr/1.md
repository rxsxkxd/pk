FROM ruby:2.7.4-bullseye

RUN sed -i '/bullseye-security/d' /etc/apt/sources.list && \
    apt-get update -qq && \
    apt-get install -y --no-install-recommends curl ca-certificates gnupg \
    && apt-get install -y --no-install-recommends nodejs wget default-mysql-client \
    && mkdir /api \
    && rm -rf /var/lib/apt/lists/*

ENV SSL_CERT_FILE=/path/to/cacert.pem を Dockerfile に1行足すだけで、gem も bundle も rails もすべて解決します。
SSL_CERT_FILE は bundle もrailsもこの環境変数を利用している？
はい、bundle（Bundler）も rails（Railsアプリ内の通信）も、すべてこの SSL_CERT_FILE 環境変数を利用しています。
